МДК 01.01/01.02 – Функция определения вида треугольника
cpp
enum TriangleType { RECTANGULAR, ACUTE, OBTUSE, INVALID };
TriangleType getTriangleType(double a, double b, double c) {
    if (a+b<=c || a+c<=b || b+c<=a) return INVALID;
    double a2=a*a, b2=b*b, c2=c*c;
    if (fabs(a2+b2-c2)<1e-9 || fabs(a2+c2-b2)<1e-9 || fabs(b2+c2-a2)<1e-9) return RECTANGULAR;
    if (a2+b2>c2 && a2+c2>b2 && b2+c2>a2) return ACUTE;
    return OBTUSE;
}
