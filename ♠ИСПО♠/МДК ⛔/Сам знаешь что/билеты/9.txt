МДК 01.04 – Функции countwords, reversestring, isstringpalindrome
cpp
int countwords(const char* str) {
    int cnt = 0;
    bool inWord = false;
    for (; *str; str++) {
        if (isspace(*str)) inWord = false;
        else if (!inWord) { inWord = true; cnt++; }
    }
    return cnt;
}

void reversestring(char* str) {
    int len = strlen(str);
    for (int i = 0; i < len/2; i++) swap(str[i], str[len-1-i]);
}

int isstringpalindrome(const char* str) {
    int len = strlen(str);
    for (int i = 0; i < len/2; i++)
        if (str[i] != str[len-1-i]) return 0;
    return 1;
}
