МДК 01.01/01.02 – Стоимость оборудования с амортизацией
Код:

cpp
#include <fstream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    ifstream fin("prices.txt");
    vector<double> prices;
    double p;
    fin >> p; // процент амортизации
    double val;
    while (fin >> val) prices.push_back(val);
    double totalCost = 0;
    for (int year = 0; year < prices.size(); year++) {
        double current = prices[year];
        int yearsOld = prices.size() - year;
        current *= pow(1 - p/100, yearsOld);
        totalCost += current;
    }
    cout << "Общая стоимость с учётом амортизации: " << totalCost << endl;
    return 0;
}
