МДК 01.01/01.02 – Циклический сдвиг массива

void shift(int arr[], int n, int k) {
    k %= n;
    if (k > 0) { // вправо
        reverse(arr, arr+n);
        reverse(arr, arr+k);
        reverse(arr+k, arr+n);
    } else if (k < 0) { // влево
        k = -k;
        reverse(arr, arr+n);
        reverse(arr, arr+n-k);
        reverse(arr+n-k, arr+n);
    }
}
