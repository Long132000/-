МДК 01.01/01.02 – Выдача суммы минимальным количеством купюр
cpp
vector<int> denominations = {5000,2000,1000,500,200,100,50};
vector<int> count(denominations.size());
int S; cin >> S;
for (size_t i=0; i<denominations.size(); i++) {
    count[i] = S / denominations[i];
    S %= denominations[i];
}
if (S != 0) throw runtime_error("Cannot pay exact sum");
