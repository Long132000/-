МДК 01.01/01.02 – Код программы для параллелепипеда (класс, ввод из файла)
Код:

cpp
#include <fstream>
#include <iostream>
#include <cmath>
using namespace std;

class Parallelepiped {
    double a, b, c;
public:
    Parallelepiped(double a, double b, double c) : a(a), b(b), c(c) {}
    double diagonal() { return sqrt(a*a + b*b + c*c); }
    double surfaceArea() { return 2*(a*b + a*c + b*c); }
    double volume() { return a*b*c; }
    void print() {
        cout << "a=" << a << " b=" << b << " c=" << c
             << " diagonal=" << diagonal()
             << " area=" << surfaceArea()
             << " volume=" << volume() << endl;
    }
};

int main() {
    ifstream fin("input.txt");
    double a,b,c;
    while (fin >> a >> b >> c) {
        Parallelepiped p(a,b,c);
        p.print();
    }
    return 0;
}
