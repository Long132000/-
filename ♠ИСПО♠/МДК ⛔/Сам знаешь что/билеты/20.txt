МДК 01.01/01.02 – Класс для обработки строк (последнее вхождение символа)
cpp
class StringUtils {
public:
    static int lastIndexOf(const string& str, char ch) {
        for (int i = str.length()-1; i >= 0; i--) if (str[i]==ch) return i;
        return -1;
    }
    static string afterLast(const string& str, char ch) {
        int pos = lastIndexOf(str, ch);
        if (pos == -1) return "";
        return str.substr(pos+1);
    }
    static string afterLastN(const string& str, char ch, int n) {
        string tail = afterLast(str, ch);
        return tail.substr(0, n);
    }
};
