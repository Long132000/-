МДК 01.01/01.02 – Вставка в массив на указанную позицию
cpp
void insert(int arr[], int& size, int value, int pos) {
    if (pos<0 || pos>size) throw out_of_range("Invalid position");
    for (int i = size; i > pos; i--) arr[i] = arr[i-1];
    arr[pos] = value;
    size++;
}
