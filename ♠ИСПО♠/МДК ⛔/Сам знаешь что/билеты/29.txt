МДК 01.01/01.02 – Проверка орфографии (словарь)
cpp
set<string> dictionary; // загрузить из файла
string text; getline(cin, text);
istringstream iss(text);
string word;
while (iss >> word) {
    if (dictionary.find(word) == dictionary.end())
        cout << "[" << word << "] "; // подчеркнуть
    else cout << word << " ";
}
