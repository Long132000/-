int fontSize = 16;    // размер шрифта

// Функция для установки цвета консоли
void setColor(int bg, int text) {
    HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);
    SetConsoleTextAttribute(hConsole, (bg << 4) | text);
}

}

// Первая форма - игра "Угадай число"
void runGame() {
    srand(time(NULL));
    int secretNumber = rand() % 100 + 1;
    int attempts = 0;
    int guess;
    time_t startTime = time(NULL);
    time_t currentTime;
    
    cout << "========== УГАДАЙ ЧИСЛО ==========" << endl;
    cout << "Я загадал число от 1 до 100." << endl;
    cout << "У тебя есть 2 минуты! Удачи!" << endl;
    cout << "===================================" << endl;
    
    while (true) {
        currentTime = time(NULL);
        int elapsedSeconds = (int)(currentTime - startTime);
        
        if (elapsedSeconds >= 120) {
            cout << "\nВремя вышло! Загаданное число было: " << secretNumber << endl;
            break;
        }
        
        cout << "\nОсталось времени: " << (120 - elapsedSeconds) << " сек" << endl;
        cout << "Введите число (или 0 для выхода): ";
        cin >> guess;
        
        if (guess == 0) {
            cout << "Игра прервана." << endl;
            break;
        }
        
        attempts++;
        
        if (guess < secretNumber) {
            cout << "БОЛЬШЕ!" << endl;
        } else if (guess > secretNumber) {
            cout << "МЕНЬШЕ!" << endl;
        } else {
            cout << "\nПОЗДРАВЛЯЮ! Ты угадал число " << secretNumber << " за " << attempts << " попыток!" << endl;
            break;
        }
    }
    
    cout << "\nИгра окончена." << endl;
}

int main() {
    // Устанавливаем кодировку для русского языка

        cout << "0 - Выход" << endl;
        cout << "==================================" << endl;

            case 1:
                runGame();
                break;

                break;
            case 0:
                cout << "До свидания!" << endl;
                break;
            default:
                cout << "Неверный выбор!" << endl;
        }
    } while (choice != 0);
    
    return 0;
}

package com.example.diagonal;

import androidx.appcompat.app.AppCompatActivity;
import android.graphics.Color;
import android.os.Bundle;
import android.widget.ArrayAdapter;
import android.widget.EditText;
import android.widget.ListView;
import java.util.ArrayList;

public class MainActivity extends AppCompatActivity {
    private EditText etA, etB, etC;
    private ListView listView;
    private ArrayList<String> items;
    private ArrayAdapter<String> adapter;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etA = findViewById(R.id.etA);
        etB = findViewById(R.id.etB);
        etC = findViewById(R.id.etC);
        listView = findViewById(R.id.listView);
        items = new ArrayList<>();
        adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, items);
        listView.setAdapter(adapter);

        findViewById(R.id.btnCalculate).setOnClickListener(v -> {
            String sA = etA.getText().toString();
            String sB = etB.getText().toString();
            String sC = etC.getText().toString();
            
            if (sA.isEmpty() || sB.isEmpty() || sC.isEmpty()) return;
            
            double a = Double.parseDouble(sA);
            double b = Double.parseDouble(sB);
            double c = Double.parseDouble(sC);
            double diagonal = Math.sqrt(a * a + b * b + c * c);
            
            items.add("a = " + a + " b = " + b + " c = " + c + " d = " + String.format("%.2f", diagonal));
            adapter.notifyDataSetChanged();
            
            etA.setText("");
            etB.setText("");
            etC.setText("");
        });

        findViewById(R.id.btnClear).setOnClickListener(v -> {
            items.clear();
            adapter.notifyDataSetChanged();
        });

        findViewById(R.id.btnTheme1).setOnClickListener(v -> {
            int textColor = Color.parseColor("#FF5722");
            int bgColor = Color.parseColor("#FFF3E0");
            applyTheme(textColor, bgColor);
        });

        findViewById(R.id.btnTheme2).setOnClickListener(v -> {
            int textColor = Color.parseColor("#4CAF50");
            int bgColor = Color.parseColor("#E8F5E9");
            applyTheme(textColor, bgColor);
        });
    }
    
    private void applyTheme(int textColor, int bgColor) {
        etA.setTextColor(textColor);
        etA.setBackgroundColor(bgColor);
        etB.setTextColor(textColor);
        etB.setBackgroundColor(bgColor);
        etC.setTextColor(textColor);
        etC.setBackgroundColor(bgColor);
    }
}

#include <windows.h>
#include <iostream>
using namespace std;

int main() {
    // 1. Время работы системы
    ULONGLONG ms = GetTickCount64();
    ULONGLONG seconds = ms / 1000;
    ULONGLONG days = seconds / 86400;
    seconds %= 86400;
    ULONGLONG hours = seconds / 3600;
    seconds %= 3600;
    ULONGLONG minutes = seconds / 60;
    seconds %= 60;

    cout << "Время работы ОС: " << days << " дней, " << hours << " часов, "
         << minutes << " минут, " << seconds << " секунд" << endl;

    // 2. MessageBox с текущим временем и датой
    SYSTEMTIME st;
    GetLocalTime(&st);
    char timeMsg[200];
    sprintf_s(timeMsg, sizeof(timeMsg),
              "Текущее время: %02d:%02d:%02d\nДата: %02d.%02d.%04d",
              st.wHour, st.wMinute, st.wSecond, st.wDay, st.wMonth, st.wYear);
    MessageBoxA(NULL, timeMsg, "Время и дата", MB_OK);

    // 3. Имя компьютера и пользователя
    char compName[256];
    DWORD size = sizeof(compName);
    GetComputerNameA(compName, &size);
    cout << "Имя компьютера: " << compName << endl;

    char userName[256];
    size = sizeof(userName);
    GetUserNameA(userName, &size);
    cout << "Имя пользователя: " << userName << endl;

    return 0;
}
