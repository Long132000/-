cpp
#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <iomanip>
#include <ctime>
using namespace std;

// Структура для хранения результата заплыва
struct SwimResult {
    char date[11];        // дата в формате ДД.ММ.ГГГГ
    char surname[50];     // фамилия спортсмена
    int distance;         // длина дистанции (м)
    double time;          // время заплыва (сек)
};

class SwimmingCompetition {
private:
    string filename;
    
    string getCurrentDate() {
        time_t now = time(nullptr);
        tm* t = localtime(&now);
        char buf[11];
        sprintf_s(buf, "%02d.%02d.%04d", t->tm_mday, t->tm_mon + 1, t->tm_year + 1900);
        return string(buf);
    }
    
public:
    SwimmingCompetition(const string& fname) : filename(fname) {}
    
    // Добавление результата в файл
    void addResult(const string& surname, int distance, double time) {
        ofstream fout(filename, ios::binary | ios::app);
        if (!fout.is_open()) {
            cout << "Ошибка открытия файла!" << endl;
            return;
        }
        
        SwimResult result;
        strcpy_s(result.date, getCurrentDate().c_str());
        strcpy_s(result.surname, surname.c_str());
        result.distance = distance;
        result.time = time;
        
        fout.write((char*)&result, sizeof(SwimResult));
        fout.close();
        
        cout << "Результат добавлен!" << endl;
    }
    
    // Чтение всех результатов из файла
    vector<SwimResult> getAllResults() {
        vector<SwimResult> results;
        ifstream fin(filename, ios::binary);
        if (!fin.is_open()) return results;
        
        SwimResult r;
        while (fin.read((char*)&r, sizeof(SwimResult))) {
            results.push_back(r);
        }
        fin.close();
        return results;
    }
    
    // Вывод таблицы результатов
    void printTable() {
        vector<SwimResult> results = getAllResults();
        
        if (results.empty()) {
            cout << "Нет данных для отображения" << endl;
            return;
        }
        
        cout << "\n========================================================================" << endl;
        cout << "|         Дата         |    Фамилия    | Дистанция (м) | Время (сек) |" << endl;
        cout << "========================================================================" << endl;
        
        for (const auto& r : results) {
            cout << "| " << setw(20) << left << r.date
                 << "| " << setw(14) << left << r.surname
                 << "| " << setw(14) << right << r.distance
                 << "| " << setw(12) << right << fixed << setprecision(2) << r.time << " |" << endl;
        }
        cout << "========================================================================" << endl;
    }
};

// Простая консольная "вторая форма" (имитация)
void showSecondForm(SwimmingCompetition& comp) {
    cout << "\n========== ВТОРАЯ ФОРМА - ТАБЛИЦА РЕЗУЛЬТАТОВ ==========" << endl;
    comp.printTable();
    cout << "==========================================================" << endl;
}

int main() {
    SwimmingCompetition comp("swim_results.bin");
    
    int choice;
    do {
        cout << "\n=== Соревнования по плаванию ===" << endl;
        cout << "1 - Добавить результат" << endl;
        cout << "2 - Показать таблицу (вторая форма)" << endl;
        cout << "0 - Выход" << endl;
        cout << "Выбор: ";
        cin >> choice;
        cin.ignore();
        
        switch (choice) {
            case 1: {
                string surname;
                int distance;
                double time;
                
                cout << "Фамилия спортсмена: ";
                getline(cin, surname);
                cout << "Дистанция (м): ";
                cin >> distance;
                cout << "Время заплыва (сек): ";
                cin >> time;
                
                comp.addResult(surname, distance, time);
                break;
            }
            case 2:
                showSecondForm(comp);
                break;
            case 0:
                cout << "До свидания!" << endl;
                break;
            default:
                cout << "Неверный выбор!" << endl;
        }
    } while (choice != 0);
    
    return 0;
}

package com.example.productthree;

import androidx.appcompat.app.AppCompatActivity;
import android.graphics.Color;
import android.os.Bundle;
import android.view.View;
import android.widget.ArrayAdapter;
import android.widget.EditText;
import android.widget.ListView;
import java.util.ArrayList;

public class MainActivity extends AppCompatActivity {
    private EditText et1, et2, et3;
    private ListView listView;
    private ArrayList<String> items;
    private ArrayAdapter<String> adapter;
    private View rootView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        rootView = findViewById(android.R.id.content);

        et1 = findViewById(R.id.et1);
        et2 = findViewById(R.id.et2);
        et3 = findViewById(R.id.et3);
        listView = findViewById(R.id.listView);
        items = new ArrayList<>();
        adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, items);
        listView.setAdapter(adapter);

        findViewById(R.id.btnAdd).setOnClickListener(v -> {
            String s1 = et1.getText().toString();
            String s2 = et2.getText().toString();
            String s3 = et3.getText().toString();
            if (s1.isEmpty() || s2.isEmpty() || s3.isEmpty()) return;
            
            int a = Integer.parseInt(s1);
            int b = Integer.parseInt(s2);
            int c = Integer.parseInt(s3);
            int product = a * b * c;
            
            items.add(product + "=" + a + "*" + b + "*" + c);
            adapter.notifyDataSetChanged();
            
            et1.setText("");
            et2.setText("");
            et3.setText("");
        });

        findViewById(R.id.btnClear).setOnClickListener(v -> {
            items.clear();
            adapter.notifyDataSetChanged();
        });

        findViewById(R.id.btnRed).setOnClickListener(v -> rootView.setBackgroundColor(Color.RED));
        findViewById(R.id.btnBlue).setOnClickListener(v -> rootView.setBackgroundColor(Color.BLUE));
    }
}
МДК 01.04 – Работа с файлом (дублирование описателя) – повтор из билета 11
cpp
#include <windows.h>
#include <iostream>
using namespace std;

int main() {
    // 1. Создаём файл
    HANDLE hFile1 = CreateFileA(
        "test_duplicate.txt",
        GENERIC_WRITE,
        0,
        NULL,
        CREATE_ALWAYS,
        FILE_ATTRIBUTE_NORMAL,
        NULL
    );
    if (hFile1 == INVALID_HANDLE_VALUE) {
        cout << "Ошибка создания файла" << endl;
        return 1;
    }
    cout << "Файл создан. Handle: " << hFile1 << endl;

    // 2. Дублируем описатель
    HANDLE hFile2;
    DuplicateHandle(
        GetCurrentProcess(),
        hFile1,
        GetCurrentProcess(),
        &hFile2,
        0,
        FALSE,
        DUPLICATE_SAME_ACCESS
    );
    cout << "Дубликат handle: " << hFile2 << endl;

    // 3. Запись через оба описателя
    DWORD written;
    char data1[] = "Запись через первый описатель\n";
    WriteFile(hFile1, data1, sizeof(data1) - 1, &written, NULL);
    cout << "Записано через первый: " << written << " байт" << endl;

    char data2[] = "Запись через второй описатель\n";
    WriteFile(hFile2, data2, sizeof(data2) - 1, &written, NULL);
    cout << "Записано через второй: " << written << " байт" << endl;

    // 4. Закрываем первый описатель
    CloseHandle(hFile1);
    cout << "Первый описатель закрыт" << endl;

    // 5. Проверяем запись через второй описатель
    char data3[] = "Запись после закрытия первого\n";
    WriteFile(hFile2, data3, sizeof(data3) - 1, &written, NULL);
    cout << "Записано через второй после закрытия первого: " << written << " байт" << endl;

    // 6. Закрываем второй описатель
    CloseHandle(hFile2);
    cout << "Второй описатель закрыт. Файл закрыт." << endl;

    return 0;
}
