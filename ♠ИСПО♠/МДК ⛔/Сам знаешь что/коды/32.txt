#include <iostream>
#include <fstream>
#include <vector>
#include <iomanip>
#include <string>
#include <ctime>
using namespace std;

// Структура для хранения давления
struct BloodPressure {
    char date[11];        // дата измерения ДД.ММ.ГГГГ
    int systolic;         // систолическое давление (верхнее)
    int diastolic;        // диастолическое давление (нижнее)
};

class PatientMonitor {
private:
    string filename;
    
    string getCurrentDate() {
        time_t now = time(nullptr);
        tm* t = localtime(&now);
        char buf[11];
        sprintf_s(buf, "%02d.%02d.%04d", t->tm_mday, t->tm_mon + 1, t->tm_year + 1900);
        return string(buf);
    }
    
public:
    PatientMonitor(const string& fname) : filename(fname) {}
    
    // Добавление измерения давления
    void addMeasurement(int systolic, int diastolic) {
        ofstream fout(filename, ios::binary | ios::app);
        if (!fout.is_open()) {
            cout << "Ошибка открытия файла!" << endl;
            return;
        }
        
        BloodPressure bp;
        strcpy_s(bp.date, getCurrentDate().c_str());
        bp.systolic = systolic;
        bp.diastolic = diastolic;
        
        fout.write((char*)&bp, sizeof(BloodPressure));
        fout.close();
        
        cout << "Измерение добавлено!" << endl;
    }
    
    // Чтение всех измерений
    vector<BloodPressure> getAllMeasurements() {
        vector<BloodPressure> measurements;
        ifstream fin(filename, ios::binary);
        if (!fin.is_open()) return measurements;
        
        BloodPressure bp;
        while (fin.read((char*)&bp, sizeof(BloodPressure))) {
            measurements.push_back(bp);
        }
        fin.close();
        return measurements;
    }
    
    // Вывод списка измерений (вторая форма)
    void printList() {
        vector<BloodPressure> measurements = getAllMeasurements();
        
        if (measurements.empty()) {
            cout << "Нет данных для отображения" << endl;
            return;
        }
        
        cout << "\n========================================================" << endl;
        cout << "|         Дата         | Систолическое | Диастолическое |" << endl;
        cout << "========================================================" << endl;
        
        for (const auto& bp : measurements) {
            cout << "| " << setw(20) << left << bp.date
                 << "| " << setw(14) << right << bp.systolic
                 << "| " << setw(15) << right << bp.diastolic << " |" << endl;
        }
        cout << "========================================================" << endl;
        
        // Анализ давления
        cout << "\n=== Анализ последнего измерения ===" << endl;
        const auto& last = measurements.back();
        cout << "Дата: " << last.date << endl;
        cout << "Давление: " << last.systolic << "/" << last.diastolic << " мм рт.ст." << endl;
        
        if (last.systolic < 120 && last.diastolic < 80) {
            cout << "Оценка: Нормальное давление" << endl;
        } else if (last.systolic < 130 && last.diastolic < 85) {
            cout << "Оценка: Повышенное нормальное" << endl;
        } else if (last.systolic < 140 && last.diastolic < 90) {
            cout << "Оценка: Умеренная гипертензия" << endl;
        } else if (last.systolic < 160 && last.diastolic < 100) {
            cout << "Оценка: Умеренная гипертензия (II степень)" << endl;
        } else {
            cout << "Оценка: Тяжёлая гипертензия! Требуется консультация врача!" << endl;
        }
    }
};

// Вторая форма (имитация)
void showSecondForm(PatientMonitor& monitor) {
    cout << "\n========== ВТОРАЯ ФОРМА - СПИСОК ИЗМЕРЕНИЙ ==========" << endl;
    monitor.printList();
    cout << "========================================================" << endl;
}

int main() {
    PatientMonitor monitor("blood_pressure.bin");
    
    int choice;
    do {
        cout << "\n=== Мониторинг давления пациента ===" << endl;
        cout << "1 - Добавить измерение" << endl;
        cout << "2 - Показать список (вторая форма)" << endl;
        cout << "0 - Выход" << endl;
        cout << "Выбор: ";
        cin >> choice;
        
        switch (choice) {
            case 1: {
                int systolic, diastolic;
                cout << "Введите систолическое давление (верхнее): ";
                cin >> systolic;
                cout << "Введите диастолическое давление (нижнее): ";
                cin >> diastolic;
                
                if (systolic <= 0 || diastolic <= 0 || systolic < diastolic) {
                    cout << "Некорректные значения давления!" << endl;
                } else {
                    monitor.addMeasurement(systolic, diastolic);
                }
                break;
            }
            case 2:
                showSecondForm(monitor);
                break;
            case 0:
                cout << "До свидания!" << endl;
                break;
            default:
                cout << "Неверный выбор!" << endl;
        }
    } while (choice != 0);
    
    return 0;
}

package com.example.heron;

import androidx.appcompat.app.AppCompatActivity;
import android.graphics.Color;
import android.os.Bundle;
import android.widget.ArrayAdapter;
import android.widget.EditText;
import android.widget.ListView;
import android.widget.Toast;
import java.util.ArrayList;

public class MainActivity extends AppCompatActivity {
    private EditText etA, etB, etC;
    private ListView listView;
    private ArrayList<String> items;
    private ArrayAdapter<String> adapter;
    private int counter = 1;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etA = findViewById(R.id.etA);
        etB = findViewById(R.id.etB);
        etC = findViewById(R.id.etC);
        listView = findViewById(R.id.listView);
        items = new ArrayList<>();
        adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, items);
        listView.setAdapter(adapter);

        findViewById(R.id.btnCalculate).setOnClickListener(v -> {
            String sA = etA.getText().toString();
            String sB = etB.getText().toString();
            String sC = etC.getText().toString();
            
            if (sA.isEmpty() || sB.isEmpty() || sC.isEmpty()) {
                Toast.makeText(this, "Введите все стороны", Toast.LENGTH_SHORT).show();
                return;
            }
            
            double a = Double.parseDouble(sA);
            double b = Double.parseDouble(sB);
            double c = Double.parseDouble(sC);
            
            // Проверка существования треугольника
            if (a + b <= c || a + c <= b || b + c <= a) {
                Toast.makeText(this, "Треугольник с такими сторонами не существует", Toast.LENGTH_SHORT).show();
                return;
            }
            
            double p = (a + b + c) / 2;
            double area = Math.sqrt(p * (p - a) * (p - b) * (p - c));
            
            items.add(counter++ + ") a = " + a + " b = " + b + " c = " + c + " S = " + String.format("%.2f", area));
            adapter.notifyDataSetChanged();
            
            etA.setText("");
            etB.setText("");
            etC.setText("");
        });

        findViewById(R.id.btnClear).setOnClickListener(v -> {
            items.clear();
            counter = 1;
            adapter.notifyDataSetChanged();
        });

        findViewById(R.id.btnTheme1).setOnClickListener(v -> {
            int textColor = Color.parseColor("#FF5722");
            int bgColor = Color.parseColor("#FFF3E0");
            applyTheme(textColor, bgColor);
        });

        findViewById(R.id.btnTheme2).setOnClickListener(v -> {
            int textColor = Color.parseColor("#4CAF50");
            int bgColor = Color.parseColor("#E8F5E9");
            applyTheme(textColor, bgColor);
        });
    }
    
    private void applyTheme(int textColor, int bgColor) {
        etA.setTextColor(textColor);
        etA.setBackgroundColor(bgColor);
        etB.setTextColor(textColor);
        etB.setBackgroundColor(bgColor);
        etC.setTextColor(textColor);
        etC.setBackgroundColor(bgColor);
    }
}
МДК 01.04 – Работа с мьютексом (DuplicateHandle) – повтор из билета 10
cpp
#include <windows.h>
#include <iostream>
using namespace std;

int main() {
    // 1. Создаём мьютекс
    HANDLE hMutex1 = CreateMutex(NULL, FALSE, NULL);
    if (hMutex1 == NULL) {
        cout << "Ошибка создания мьютекса" << endl;
        return 1;
    }
    cout << "Мьютекс создан. Handle: " << hMutex1 << endl;

    // 2. Дублируем описатель
    HANDLE hMutex2;
    BOOL duplicated = DuplicateHandle(
        GetCurrentProcess(),
        hMutex1,
        GetCurrentProcess(),
        &hMutex2,
        0,
        FALSE,
        DUPLICATE_SAME_ACCESS
    );
    if (!duplicated) {
        cout << "Ошибка дублирования" << endl;
        CloseHandle(hMutex1);
        return 1;
    }
    cout << "Дубликат создан. Handle: " << hMutex2 << endl;

    // 3. Закрываем первый handle
    CloseHandle(hMutex1);
    cout << "Первый handle закрыт. Мьютекс всё ещё существует (через второй handle)." << endl;

    // Проверяем, что мьютекс доступен через второй handle
    DWORD waitResult = WaitForSingleObject(hMutex2, 1000);
    if (waitResult == WAIT_OBJECT_0) {
        cout << "Мьютекс успешно захвачен через второй handle" << endl;
        ReleaseMutex(hMutex2);
    } else {
        cout << "Не удалось захватить мьютекс" << endl;
    }

    // 4. Закрываем второй handle
    CloseHandle(hMutex2);
    cout << "Второй handle закрыт. Мьютекс уничтожен." << endl;

    return 0;
}
