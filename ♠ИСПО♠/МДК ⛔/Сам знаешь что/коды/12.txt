package com.example.familyyear;

import androidx.appcompat.app.AppCompatActivity;
import android.graphics.Color;
import android.os.Bundle;
import android.view.View;
import android.widget.ArrayAdapter;
import android.widget.EditText;
import android.widget.ListView;
import java.util.ArrayList;

public class MainActivity extends AppCompatActivity {
    private EditText etSurname, etYear;
    private ListView listView;
    private ArrayList<String> items;
    private ArrayAdapter<String> adapter;
    private View rootView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        rootView = findViewById(android.R.id.content);

        etSurname = findViewById(R.id.etSurname);
        etYear = findViewById(R.id.etYear);
        listView = findViewById(R.id.listView);
        items = new ArrayList<>();
        adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, items);
        listView.setAdapter(adapter);

        findViewById(R.id.btnAdd).setOnClickListener(v -> {
            String surname = etSurname.getText().toString().trim();
            String year = etYear.getText().toString().trim();
            if (!surname.isEmpty() && !year.isEmpty()) {
                items.add("Фамилия: " + surname + " Год рождения: " + year);
                adapter.notifyDataSetChanged();
            }
        });

        findViewById(R.id.btnClear).setOnClickListener(v -> {
            items.clear();
            adapter.notifyDataSetChanged();
        });

        findViewById(R.id.btnGreen).setOnClickListener(v -> rootView.setBackgroundColor(Color.GREEN));
        findViewById(R.id.btnBlue).setOnClickListener(v -> rootView.setBackgroundColor(Color.BLUE));
    }
}
МДК 01.04 – Наследование описателя мьютекса дочерним процессом
Родительский процесс (parent.cpp):

cpp
#include <windows.h>
#include <iostream>
#include <string>
using namespace std;

int main() {
    // Создаём мьютекс с атрибутом наследования
    SECURITY_ATTRIBUTES sa;
    sa.nLength = sizeof(sa);
    sa.lpSecurityDescriptor = NULL;
    sa.bInheritHandle = TRUE;   // РАЗРЕШАЕМ наследование

    HANDLE hMutex = CreateMutex(&sa, FALSE, NULL);
    if (hMutex == NULL) {
        cout << "Ошибка создания мьютекса" << endl;
        return 1;
    }
    cout << "Мьютекс создан. Handle: " << hMutex << endl;

    // Готовим командную строку для дочернего процесса (передаём handle как число)
    char cmdLine[256];
    sprintf_s(cmdLine, sizeof(cmdLine), "child.exe %p", hMutex);

    // Запускаем дочерний процесс
    STARTUPINFOA si;
    PROCESS_INFORMATION pi;
    ZeroMemory(&si, sizeof(si));
    si.cb = sizeof(si);
    ZeroMemory(&pi, sizeof(pi));

    BOOL created = CreateProcessA(
        NULL,           // имя исполняемого файла (берётся из cmdLine)
        cmdLine,        // командная строка
        NULL,           // атрибуты процесса
        NULL,           // атрибуты потока
        TRUE,           // НАСЛЕДОВАТЬ описатели!
        0,              // флаги создания
        NULL,           // переменные окружения
        NULL,           // текущая директория
        &si,
        &pi
    );

    if (!created) {
        cout << "Ошибка запуска дочернего процесса" << endl;
        CloseHandle(hMutex);
        return 1;
    }

    cout << "Дочерний процесс запущен. PID: " << pi.dwProcessId << endl;

    // Ждём завершения дочернего процесса
    WaitForSingleObject(pi.hProcess, INFINITE);

    // Закрываем handle'ы
    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);
    CloseHandle(hMutex);

    cout << "Родитель завершён." << endl;
    return 0;
}
Дочерний процесс (child.cpp):

cpp
#include <windows.h>
#include <iostream>
#include <string>
using namespace std;

int main(int argc, char* argv[]) {
    if (argc < 2) {
        cout << "Дочерний процесс: не передан handle мьютекса" << endl;
        return 1;
    }

    // Преобразуем строку в handle
    HANDLE hMutex = (HANDLE)strtoull(argv[1], NULL, 16);  // для 64-битных систем
    // Или просто: HANDLE hMutex = (HANDLE)atoi(argv[1]); для 32-битных

    cout << "Дочерний процесс получил handle: " << hMutex << endl;

    // Используем мьютекс
    DWORD waitResult = WaitForSingleObject(hMutex, 5000);
    if (waitResult == WAIT_OBJECT_0) {
        cout << "Дочерний процесс захватил мьютекс!" << endl;
        // Имитируем работу
        Sleep(1000);
        ReleaseMutex(hMutex);
        cout << "Дочерний процесс освободил мьютекс" << endl;
    } else {
        cout << "Дочерний процесс НЕ смог захватить мьютекс" << endl;
    }

    cout << "Дочерний процесс завершён." << endl;
    return 0;
}
