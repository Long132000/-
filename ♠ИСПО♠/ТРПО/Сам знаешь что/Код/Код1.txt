import sqlite3
import datetime

def init_db():
    conn = sqlite3.connect('tourism.db')
    cur = conn.cursor()
    cur.executescript('''
        CREATE TABLE IF NOT EXISTS Country (
            id INTEGER PRIMARY KEY, name TEXT NOT NULL
        );
        CREATE TABLE IF NOT EXISTS Manager (
            id INTEGER PRIMARY KEY, full_name TEXT NOT NULL
        );
        CREATE TABLE IF NOT EXISTS Tour (
            id INTEGER PRIMARY KEY, name TEXT, country_id INTEGER,
            start_date DATE, end_date DATE, seats INTEGER, manager_id INTEGER,
            FOREIGN KEY(country_id) REFERENCES Country(id),
            FOREIGN KEY(manager_id) REFERENCES Manager(id)
        );
        CREATE TABLE IF NOT EXISTS Client (
            id INTEGER PRIMARY KEY, full_name TEXT, age INTEGER,
            passport_data TEXT, requesting_authority TEXT
        );
        CREATE TABLE IF NOT EXISTS Application (
            id INTEGER PRIMARY KEY, tour_id INTEGER, client_id INTEGER,
            application_date DATE, status TEXT, organizer_response TEXT,
            FOREIGN KEY(tour_id) REFERENCES Tour(id),
            FOREIGN KEY(client_id) REFERENCES Client(id)
        );
    ''')
    conn.commit()
    conn.close()

def create_application(tour_id, client_data):
    conn = sqlite3.connect('tourism.db')
    cur = conn.cursor()
    cur.execute('''INSERT INTO Client (full_name, age, passport_data, requesting_authority)
                   VALUES (?,?,?,?)''',
                (client_data['fio'], client_data['age'], client_data['passport'], client_data['authority']))
    client_id = cur.lastrowid
    cur.execute('''INSERT INTO Application (tour_id, client_id, application_date, status)
                   VALUES (?,?,?,?)''',
                (tour_id, client_id, datetime.date.today(), 'new'))
    conn.commit()
    conn.close()

def export_applications_to_file(tour_id, filename):
    conn = sqlite3.connect('tourism.db')
    cur = conn.cursor()
    cur.execute('''
        SELECT a.id, c.full_name, c.age, a.application_date, a.status
        FROM Application a JOIN Client c ON a.client_id = c.id
        WHERE a.tour_id = ?
    ''', (tour_id,))
    rows = cur.fetchall()
    with open(filename, 'w', encoding='utf-8') as f:
        for row in rows:
            f.write(f"Заявка {row[0]}: {row[1]}, возраст {row[2]}, дата {row[3]}, статус {row[4]}\n")
    conn.close()

using System;
using System.Data.SQLite;
using System.IO;

class Ticket1
{
    static string connStr = "Data Source=tourism.db";

    static void CreateApplication(int tourId, string fio, int age, string passport, string authority)
    {
        using var conn = new SQLiteConnection(connStr);
        conn.Open();
        var cmd = new SQLiteCommand("INSERT INTO Client (full_name, age, passport_data, requesting_authority) VALUES (@fio, @age, @pass, @auth); SELECT last_insert_rowid();", conn);
        cmd.Parameters.AddWithValue("@fio", fio);
        cmd.Parameters.AddWithValue("@age", age);
        cmd.Parameters.AddWithValue("@pass", passport);
        cmd.Parameters.AddWithValue("@auth", authority);
        long clientId = (long)cmd.ExecuteScalar();
        cmd = new SQLiteCommand("INSERT INTO Application (tour_id, client_id, application_date, status) VALUES (@tid, @cid, date('now'), 'new')", conn);
        cmd.Parameters.AddWithValue("@tid", tourId);
        cmd.Parameters.AddWithValue("@cid", clientId);
        cmd.ExecuteNonQuery();
    }

    static void ExportApplications(int tourId, string filename)
    {
        using var conn = new SQLiteConnection(connStr);
        conn.Open();
        var cmd = new SQLiteCommand("SELECT a.id, c.full_name, c.age, a.application_date, a.status FROM Application a JOIN Client c ON a.client_id = c.id WHERE a.tour_id = @tid", conn);
        cmd.Parameters.AddWithValue("@tid", tourId);
        using var reader = cmd.ExecuteReader();
        using var sw = new StreamWriter(filename);
        while (reader.Read())
            sw.WriteLine($"Заявка {reader[0]}: {reader[1]}, возраст {reader[2]}, дата {reader[3]}, статус {reader[4]}");
    }
}
