import sqlite3
from datetime import datetime

def get_current_debt(contract_id):
    with sqlite3.connect('bank.db') as conn:
        cur = conn.cursor()
        cur.execute("SELECT remaining_debt FROM Payment WHERE contract_id=? ORDER BY payment_date DESC LIMIT 1", (contract_id,))
        row = cur.fetchone()
        if row:
            return row[0]
        cur.execute("SELECT total_amount FROM CreditContract WHERE id=?", (contract_id,))
        return cur.fetchone()[0]

def add_payment(contract_id, amount, payment_date=None):
    with sqlite3.connect('bank.db') as conn:
        cur = conn.cursor()
        current_debt = get_current_debt(contract_id)
        new_debt = max(current_debt - amount, 0)
        cur.execute('''INSERT INTO Payment (contract_id, payment_date, amount, remaining_debt)
                       VALUES (?,?,?,?)''',
                    (contract_id, payment_date or datetime.now().date(), amount, new_debt))
        cur.execute('UPDATE CreditContract SET remaining_debt=? WHERE id=?', (new_debt, contract_id))
        if new_debt == 0:
            cur.execute('''UPDATE CreditApplication SET status="closed"
                           WHERE id=(SELECT application_id FROM CreditContract WHERE id=?)''', (contract_id,))
        conn.commit()
        return new_debt

def get_payments_by_month(contract_id, year, month):
    with sqlite3.connect('bank.db') as conn:
        cur = conn.cursor()
        start_date = f"{year}-{month:02d}-01"
        if month == 12:
            end_date = f"{year+1}-01-01"
        else:
            end_date = f"{year}-{month+1:02d}-01"
        cur.execute('''SELECT payment_date, amount, remaining_debt FROM Payment
                       WHERE contract_id=? AND payment_date >= ? AND payment_date < ?
                       ORDER BY payment_date''', (contract_id, start_date, end_date))
        return cur.fetchall()

using System;
using System.Data.SQLite;
using System.Collections.Generic;

class Ticket16
{
    static string connStr = "Data Source=bank.db";

    static decimal GetCurrentDebt(int contractId)
    {
        using var conn = new SQLiteConnection(connStr);
        conn.Open();
        var cmd = new SQLiteCommand("SELECT remaining_debt FROM Payment WHERE contract_id=@cid ORDER BY payment_date DESC LIMIT 1", conn);
        cmd.Parameters.AddWithValue("@cid", contractId);
        var result = cmd.ExecuteScalar();
        if (result != DBNull.Value)
            return Convert.ToDecimal(result);
        cmd = new SQLiteCommand("SELECT total_amount FROM CreditContract WHERE id=@cid", conn);
        cmd.Parameters.AddWithValue("@cid", contractId);
        return Convert.ToDecimal(cmd.ExecuteScalar());
    }

    static decimal AddPayment(int contractId, decimal amount, DateTime? paymentDate = null)
    {
        using var conn = new SQLiteConnection(connStr);
        conn.Open();
        decimal currentDebt = GetCurrentDebt(contractId);
        decimal newDebt = Math.Max(currentDebt - amount, 0);
        var cmd = new SQLiteCommand("INSERT INTO Payment (contract_id, payment_date, amount, remaining_debt) VALUES (@cid, @pdate, @amt, @debt)", conn);
        cmd.Parameters.AddWithValue("@cid", contractId);
        cmd.Parameters.AddWithValue("@pdate", (paymentDate ?? DateTime.Now).ToString("yyyy-MM-dd"));
        cmd.Parameters.AddWithValue("@amt", amount);
        cmd.Parameters.AddWithValue("@debt", newDebt);
        cmd.ExecuteNonQuery();
        cmd = new SQLiteCommand("UPDATE CreditContract SET remaining_debt=@debt WHERE id=@cid", conn);
        cmd.Parameters.AddWithValue("@debt", newDebt);
        cmd.Parameters.AddWithValue("@cid", contractId);
        cmd.ExecuteNonQuery();
        if (newDebt == 0)
        {
            cmd = new SQLiteCommand("UPDATE CreditApplication SET status='closed' WHERE id=(SELECT application_id FROM CreditContract WHERE id=@cid)", conn);
            cmd.Parameters.AddWithValue("@cid", contractId);
            cmd.ExecuteNonQuery();
        }
        return newDebt;
    }
}
